```dataviewjs
const activeFile = app.workspace.getActiveFile();
const cachedFrontmatter = activeFile
  ? app.metadataCache.getFileCache(activeFile)?.frontmatter
  : null;
const page = dv.current() ?? {
  ...(cachedFrontmatter ?? {}),
  file: { name: activeFile?.basename ?? "订阅档案", path: activeFile?.path ?? "" }
};

const normalizePath = value => value?.path
  ?? String(value ?? "").replace(/^\[\[|\]\]$/g, "").split("|")[0];
const toDate = value => {
  if (!value) return null;
  if (value instanceof Date) return value;
  if (typeof value.toJSDate === "function") return value.toJSDate();
  const date = new Date(`${String(value).slice(0, 10)}T00:00:00`);
  return Number.isNaN(date.getTime()) ? null : date;
};
const pad = value => String(value).padStart(2, "0");
const formatDate = value => {
  const date = toDate(value);
  if (!date) return "未记录";
  return `${date.getFullYear()}.${pad(date.getMonth() + 1)}.${pad(date.getDate())}`;
};
const formatMoney = value => {
  const amount = Number(value);
  if (!Number.isFinite(amount)) return "—";
  return `¥${amount.toLocaleString("zh-CN", {
    minimumFractionDigits: Number.isInteger(amount) ? 0 : 2,
    maximumFractionDigits: Number.isInteger(amount) ? 0 : 2
  })}`;
};
const amountOrZero = value => {
  const amount = Number(value);
  return Number.isFinite(amount) && amount >= 0 ? amount : 0;
};
const cycleMonths = { "月度": 1, "季度": 3, "年度": 12 }[String(page.cycle)] ?? 1;
const monthly = amountOrZero(page.price) / cycleMonths;
const today = new Date();
today.setHours(0, 0, 0, 0);
const nextDate = toDate(page.next_date);
const days = nextDate ? Math.ceil((nextDate - today) / 86400000) : null;
const dueAction = String(page.renewal_mode) === "手动续费" ? "到期" : "扣费";
const dueCopy = days === null
  ? "日期未记录"
  : days === 0
    ? `今天${dueAction}`
    : days > 0
      ? `${days} 天后${dueAction}`
      : `已过${dueAction}日 ${Math.abs(days)} 天`;
const paymentRecords = dv.pages('"30 订阅/_续费记录"')
  .where(record => record.type === "subscription_payment" && record.voided !== true && normalizePath(record.subscription) === page.file.path)
  .array()
  .sort((left, right) => (toDate(right.payment_date)?.getTime() ?? 0) - (toDate(left.payment_date)?.getTime() ?? 0));
const recordedSpend = paymentRecords
  .reduce((total, record) => total + amountOrZero(record.amount), amountOrZero(page.historical_spend));
const hasSpendData = paymentRecords.length > 0 || amountOrZero(page.historical_spend) > 0;

const findTrigger = type => dv.container
  .closest(".markdown-preview-view, .markdown-reading-view")
  ?.querySelector(`.callout[data-callout="subscription-item-${type}-trigger"] button`);
const launch = type => {
  const trigger = findTrigger(type);
  if (trigger instanceof HTMLElement) trigger.click();
  else new Notice("操作暂时没有加载，请重新打开订阅档案");
};
const undoPayment = async record => {
  const recordFile = app.vault.getAbstractFileByPath(record.file.path);
  if (!recordFile?.extension) {
    new Notice("找不到这笔续费记录");
    return;
  }
  const confirmed = window.confirm(`撤销 ${formatDate(record.payment_date)} 的 ${formatMoney(record.amount)} 续费记录？\n\n记录会移入“_已撤销”，不会永久删除。`);
  if (!confirmed) return;
  try {
    await app.fileManager.processFrontMatter(recordFile, data => {
      data.voided = true;
      data.voided_at = new Date().toISOString();
    });
    const archiveRoot = "30 订阅/_续费记录/_已撤销";
    if (!app.vault.getAbstractFileByPath(archiveRoot)) await app.vault.createFolder(archiveRoot);
    let archivePath = `${archiveRoot}/${recordFile.name}`;
    let suffix = 2;
    while (app.vault.getAbstractFileByPath(archivePath)) {
      archivePath = `${archiveRoot}/${recordFile.basename}-${suffix}.md`;
      suffix += 1;
    }
    await app.fileManager.renameFile(recordFile, archivePath);

    const isLatest = paymentRecords[0]?.file.path === record.file.path;
    const currentNext = String(page.next_date ?? "").slice(0, 10);
    const recordNext = String(record.next_date ?? "").slice(0, 10);
    const previousDate = String(record.previous_date ?? "").slice(0, 10);
    if (isLatest && currentNext === recordNext && toDate(previousDate)) {
      await app.fileManager.processFrontMatter(activeFile, data => {
        data.next_date = previousDate;
      });
    }
    await new Promise(resolve => setTimeout(resolve, 160));
    await app.commands.executeCommandById("dataview:refresh-views");
    new Notice("续费记录已撤销，可在“_已撤销”文件夹中找回");
  } catch (error) {
    console.error(error);
    new Notice(`撤销失败：${error.message ?? error}`);
  }
};

const root = dv.container.createDiv({ cls: "subscription-item-root" });
const topLine = root.createDiv({ cls: "subscription-item-topline" });
const back = topLine.createEl("a", {
  text: "返回订阅账簿",
  cls: "internal-link subscription-item-back"
});
back.setAttr("data-href", "30 订阅/订阅主页");
back.setAttr("href", "30 订阅/订阅主页");
const header = root.createDiv({ cls: "subscription-item-header" });
const logo = header.createEl("button", {
  cls: "subscription-item-logo",
  attr: { type: "button", "aria-label": page.cover ? "更换订阅 Logo" : "添加订阅 Logo" }
});
const coverPath = normalizePath(page.cover);
const coverFile = app.vault.getAbstractFileByPath(coverPath);
if (coverFile?.extension) {
  logo.createEl("img", {
    attr: { src: app.vault.getResourcePath(coverFile), alt: String(page.name ?? page.file.name) }
  });
  logo.createSpan({ text: "更换 Logo", cls: "subscription-item-logo-edit" });
} else {
  logo.createSpan({ text: "添加 Logo", cls: "subscription-item-logo-placeholder" });
}
logo.addEventListener("click", event => {
  event.preventDefault();
  event.stopPropagation();
  launch("cover");
});

const info = header.createDiv({ cls: "subscription-item-info" });
info.createSpan({ text: "PERSONAL SUBSCRIPTION ARCHIVE", cls: "subscription-item-kicker" });
info.createEl("h1", { text: String(page.name ?? page.file.name) });
const badges = info.createDiv({ cls: "subscription-item-badges" });
badges.createSpan({ text: String(page.status ?? "订阅中"), cls: `subscription-item-status is-${String(page.status ?? "订阅中")}` });
badges.createSpan({ text: String(page.renewal_mode ?? "自动续费"), cls: "subscription-item-mode" });

const heroCost = info.createDiv({ cls: "subscription-item-hero-cost" });
heroCost.createSpan({ text: formatMoney(monthly) });
heroCost.createEl("small", { text: "/ 月" });
info.createSpan({ text: dueCopy, cls: `subscription-item-due${days !== null && days <= 7 ? " is-accent" : ""}` });

const facts = root.createDiv({ cls: "subscription-item-facts" });
for (const [label, value, className] of [
  ["预计续费价", `${formatMoney(page.price)} / ${String(page.cycle ?? "月度").replace("度", "")}`, ""],
  [String(page.renewal_mode) === "手动续费" ? "到期日期" : "下次扣费", formatDate(page.next_date), days !== null && days <= 7 ? "is-accent" : ""],
  ["距离日期", dueCopy, days !== null && days <= 7 ? "is-accent" : ""],
  ["累计支出", hasSpendData ? formatMoney(recordedSpend) : "未录入", ""]
]) {
  const fact = facts.createDiv({ cls: "subscription-item-fact" });
  fact.createSpan({ text: label });
  fact.createEl("strong", { text: value, cls: className });
}

const actions = root.createDiv({ cls: "subscription-item-actions" });
const renew = actions.createEl("button", {
  text: "记录续费",
  cls: "subscription-item-action is-primary",
  attr: { type: "button" }
});
renew.disabled = String(page.status) === "已停用";
renew.addEventListener("click", () => launch("renew"));
const edit = actions.createEl("button", {
  text: "编辑订阅",
  cls: "subscription-item-action",
  attr: { type: "button" }
});
edit.addEventListener("click", () => launch("edit"));
const toggle = actions.createEl("button", {
  text: String(page.status) === "已停用" ? "重新启用" : "停用订阅",
  cls: "subscription-item-action is-text",
  attr: { type: "button" }
});
toggle.addEventListener("click", async () => {
  if (!activeFile) return;
  const nextStatus = String(page.status) === "已停用" ? "订阅中" : "已停用";
  await app.fileManager.processFrontMatter(activeFile, data => {
    data.status = nextStatus;
  });
  await new Promise(resolve => setTimeout(resolve, 120));
  await app.commands.executeCommandById("dataview:refresh-views");
  new Notice(`订阅状态已更新为「${nextStatus}」`);
});

const history = root.createDiv({ cls: "subscription-item-history" });
const historyHeading = history.createDiv({ cls: "subscription-item-history-heading" });
const historyTitle = historyHeading.createDiv();
historyTitle.createSpan({ text: "THE RENEWAL LEDGER", cls: "subscription-item-kicker" });
historyTitle.createEl("h2", { text: "续费记录" });
historyHeading.createSpan({ text: `${paymentRecords.length} 笔`, cls: "subscription-item-history-count" });
if (paymentRecords.length) {
  const list = history.createDiv({ cls: "subscription-item-history-list" });
  for (const record of paymentRecords.slice(0, 8)) {
    const row = list.createDiv({ cls: "subscription-item-history-row" });
    row.createSpan({ text: formatDate(record.payment_date) });
    row.createEl("strong", { text: formatMoney(record.amount) });
    const controls = row.createDiv({ cls: "subscription-item-history-controls" });
    controls.createSpan({ text: record.next_date ? `延至 ${formatDate(record.next_date)}` : "日期未记录" });
    const undo = controls.createEl("button", {
      text: "撤销",
      cls: "subscription-item-history-undo",
      attr: { type: "button", "aria-label": `撤销 ${formatDate(record.payment_date)} 的续费记录` }
    });
    undo.addEventListener("click", () => undoPayment(record));
  }
} else {
  history.createDiv({ text: "还没有续费记录。第一次点击「记录续费」后，账单会出现在这里。", cls: "subscription-item-history-empty" });
}
```
