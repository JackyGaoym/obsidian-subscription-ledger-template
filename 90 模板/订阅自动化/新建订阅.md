const { Modal, Setting } = tp.obsidian;
const pad = value => String(value).padStart(2, "0");
const toISO = date => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
const addMonths = (iso, months) => {
  const [year, month, day] = iso.split("-").map(Number);
  const target = new Date(year, month - 1 + months, 1);
  const lastDay = new Date(target.getFullYear(), target.getMonth() + 1, 0).getDate();
  target.setDate(Math.min(day, lastDay));
  return toISO(target);
};
const isValidISODate = value => {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(value))) return false;
  const [year, month, day] = String(value).split("-").map(Number);
  const date = new Date(year, month - 1, day);
  return date.getFullYear() === year && date.getMonth() === month - 1 && date.getDate() === day;
};
const cycleMonths = cycle => ({ "月度": 1, "季度": 3, "年度": 12 }[cycle] ?? 1);

const openForm = () => new Promise(resolve => {
  class SubscriptionFormModal extends Modal {
    constructor(app) {
      super(app);
      this.resolved = false;
      this.values = {
        name: "",
        status: "订阅中",
        renewal_mode: "自动续费",
        cycle: "月度",
        price: "",
        next_date: addMonths(tp.date.now("YYYY-MM-DD"), 1),
        initial_mode: "none",
        historical_spend: ""
      };
    }

    finish(value) {
      if (this.resolved) return;
      this.resolved = true;
      resolve(value);
      this.close();
    }

    onOpen() {
      const { contentEl, modalEl } = this;
      modalEl.addClass("subscription-form-modal");
      contentEl.createEl("h2", { text: "新增订阅" });
      contentEl.createEl("p", {
        text: "一次填完订阅信息；Logo 可以建立档案后再添加。",
        cls: "subscription-form-intro"
      });

      new Setting(contentEl)
        .setName("订阅名称")
        .setDesc("例如 Spotify Premium")
        .addText(text => text
          .setPlaceholder("请输入名称")
          .onChange(value => { this.values.name = value; }));

      new Setting(contentEl)
        .setName("订阅状态")
        .addDropdown(dropdown => dropdown
          .addOptions({ "订阅中": "订阅中", "已停用": "已停用" })
          .setValue(this.values.status)
          .onChange(value => { this.values.status = value; }));

      new Setting(contentEl)
        .setName("续费方式")
        .addDropdown(dropdown => dropdown
          .addOptions({ "自动续费": "自动续费", "手动续费": "手动续费" })
          .setValue(this.values.renewal_mode)
          .onChange(value => { this.values.renewal_mode = value; }));

      let dateTouched = false;
      let dateInput;
      new Setting(contentEl)
        .setName("订阅周期")
        .addDropdown(dropdown => dropdown
          .addOptions({ "月度": "月度", "季度": "季度", "年度": "年度" })
          .setValue(this.values.cycle)
          .onChange(value => {
            this.values.cycle = value;
            if (!dateTouched) {
              this.values.next_date = addMonths(tp.date.now("YYYY-MM-DD"), cycleMonths(value));
              dateInput?.setValue(this.values.next_date);
            }
          }));

      new Setting(contentEl)
        .setName("预计续费价")
        .setDesc("按下次大概率支付的价格填写，不必填写官方定价")
        .addText(text => {
          text.inputEl.type = "number";
          text.inputEl.min = "0";
          text.inputEl.step = "0.01";
          text.setPlaceholder("0.00").onChange(value => { this.values.price = value; });
        });

      new Setting(contentEl)
        .setName("下次扣费 / 到期日期")
        .addText(text => {
          dateInput = text;
          text.inputEl.type = "date";
          text.setValue(this.values.next_date).onChange(value => {
            dateTouched = true;
            this.values.next_date = value;
          });
        });

      let historySetting;
      new Setting(contentEl)
        .setName("初始支出")
        .setDesc("用于让累计支出从第一天就准确")
        .addDropdown(dropdown => dropdown
          .addOptions({
            none: "暂不录入",
            payment: "把本次购买记为首笔支出",
            history: "填写此前累计支出"
          })
          .setValue(this.values.initial_mode)
          .onChange(value => {
            this.values.initial_mode = value;
            historySetting.settingEl.toggle(value === "history");
          }));

      historySetting = new Setting(contentEl)
        .setName("此前累计支出")
        .setDesc("不含以后通过“记录续费”新增的支出")
        .addText(text => {
          text.inputEl.type = "number";
          text.inputEl.min = "0";
          text.inputEl.step = "0.01";
          text.setPlaceholder("0.00").onChange(value => { this.values.historical_spend = value; });
        });
      historySetting.settingEl.hide();

      new Setting(contentEl)
        .setClass("subscription-form-actions")
        .addButton(button => button
          .setButtonText("取消")
          .onClick(() => this.finish(null)))
        .addButton(button => button
          .setButtonText("建立订阅")
          .setCta()
          .onClick(() => {
            const name = this.values.name.trim();
            const price = Number(this.values.price);
            const historicalSpend = Number(this.values.historical_spend || 0);
            if (!name) {
              new Notice("请填写订阅名称");
              return;
            }
            if (!Number.isFinite(price) || price < 0) {
              new Notice("预计续费价需要填写有效数字");
              return;
            }
            if (!isValidISODate(this.values.next_date)) {
              new Notice("请填写有效的下次扣费或到期日期");
              return;
            }
            if (this.values.initial_mode === "history" && (!Number.isFinite(historicalSpend) || historicalSpend < 0)) {
              new Notice("此前累计支出需要填写有效数字");
              return;
            }
            this.finish({
              ...this.values,
              name,
              price,
              historical_spend: this.values.initial_mode === "history" ? historicalSpend : null
            });
          }));
    }

    onClose() {
      this.contentEl.empty();
      if (!this.resolved) {
        this.resolved = true;
        resolve(null);
      }
    }
  }

  new SubscriptionFormModal(app).open();
});

const values = await openForm();
if (!values) return;

const safeName = values.name.replace(/[\\/:*?"<>|]/g, "-");
const root = "30 订阅/项目";
let folder = `${root}/${safeName}`;
let suffix = 2;
while (app.vault.getAbstractFileByPath(folder)) {
  folder = `${root}/${safeName} ${suffix}`;
  suffix += 1;
}
const archiveName = folder.split("/").pop();
const itemPath = `${folder}/${archiveName}.md`;
const templatePath = "90 模板/订阅自动化/订阅档案母版.md";

const createInitialPayment = async file => {
  if (values.initial_mode !== "payment") return;
  const paymentDate = tp.date.now("YYYY-MM-DD");
  const recordRoot = "30 订阅/_续费记录";
  const yearFolder = `${recordRoot}/${paymentDate.slice(0, 4)}`;
  if (!app.vault.getAbstractFileByPath(recordRoot)) await app.vault.createFolder(recordRoot);
  if (!app.vault.getAbstractFileByPath(yearFolder)) await app.vault.createFolder(yearFolder);
  const stamp = tp.date.now("HHmmss");
  let recordPath = `${yearFolder}/${safeName} · ${paymentDate} · ${stamp}.md`;
  let recordSuffix = 2;
  while (app.vault.getAbstractFileByPath(recordPath)) {
    recordPath = `${yearFolder}/${safeName} · ${paymentDate} · ${stamp}-${recordSuffix}.md`;
    recordSuffix += 1;
  }
  const content = [
    "---",
    "type: subscription_payment",
    `subscription: ${JSON.stringify(`[[${file.path}|${values.name}]]`)}`,
    `payment_date: ${paymentDate}`,
    `amount: ${values.price}`,
    `cycle: ${JSON.stringify(values.cycle)}`,
    "previous_date:",
    `next_date: ${values.next_date}`,
    "source: initial_purchase",
    "---",
    ""
  ].join("\n");
  await app.vault.create(recordPath, content);
};

try {
  if (!app.vault.getAbstractFileByPath("30 订阅")) await app.vault.createFolder("30 订阅");
  if (!app.vault.getAbstractFileByPath(root)) await app.vault.createFolder(root);
  await app.vault.createFolder(folder);
  const template = app.vault.getAbstractFileByPath(templatePath);
  if (!template?.extension) throw new Error("找不到订阅档案母版");
  let content = await app.vault.read(template);
  const replacements = {
    "{{NAME_YAML}}": JSON.stringify(values.name),
    "{{STATUS_YAML}}": JSON.stringify(values.status),
    "{{RENEWAL_MODE_YAML}}": JSON.stringify(values.renewal_mode),
    "{{CYCLE_YAML}}": JSON.stringify(values.cycle),
    "{{PRICE}}": String(values.price),
    "{{NEXT_DATE}}": values.next_date,
    "{{HISTORICAL_SPEND}}": values.historical_spend === null ? "null" : String(values.historical_spend)
  };
  for (const [token, value] of Object.entries(replacements)) {
    content = content.split(token).join(value);
  }
  const file = await app.vault.create(itemPath, content);
  try {
    await createInitialPayment(file);
  } catch (error) {
    console.error(error);
    new Notice("订阅已建立，但首笔支出没有写入，请稍后手动记录续费");
  }
  new Notice(`「${values.name}」已经加入订阅账簿`);
  await app.workspace.getLeaf(false).openFile(file);
} catch (error) {
  console.error(error);
  new Notice(`新增失败：${error.message ?? error}`);
}
