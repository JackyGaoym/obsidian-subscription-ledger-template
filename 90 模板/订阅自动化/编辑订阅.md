<%*
const { Modal, Setting } = tp.obsidian;
const activeFile = app.workspace.getActiveFile();
const frontmatter = activeFile
  ? app.metadataCache.getFileCache(activeFile)?.frontmatter
  : null;
if (!activeFile || frontmatter?.type !== "subscription") {
  new Notice("请先打开一份订阅档案");
  return;
}

const isValidISODate = value => {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(value))) return false;
  const [year, month, day] = String(value).split("-").map(Number);
  const date = new Date(year, month - 1, day);
  return date.getFullYear() === year && date.getMonth() === month - 1 && date.getDate() === day;
};

const openForm = () => new Promise(resolve => {
  class EditSubscriptionModal extends Modal {
    constructor(app) {
      super(app);
      this.resolved = false;
      this.values = {
        name: String(frontmatter.name ?? activeFile.basename),
        status: String(frontmatter.status) === "已停用" ? "已停用" : "订阅中",
        renewal_mode: String(frontmatter.renewal_mode) === "手动续费" ? "手动续费" : "自动续费",
        cycle: ["月度", "季度", "年度"].includes(String(frontmatter.cycle)) ? String(frontmatter.cycle) : "月度",
        price: String(frontmatter.price ?? ""),
        next_date: String(frontmatter.next_date ?? "").slice(0, 10),
        historical_spend: frontmatter.historical_spend == null ? "" : String(frontmatter.historical_spend)
      };
    }

    finish(value) {
      if (this.resolved) return;
      this.resolved = true;
      resolve(value);
      this.close();
    }

    onOpen() {
      const { contentEl, modalEl } = this;
      modalEl.addClass("subscription-form-modal");
      contentEl.createEl("h2", { text: "编辑订阅" });
      contentEl.createEl("p", {
        text: "这里修改的是订阅规则；单次实际支出仍由“记录续费”负责。",
        cls: "subscription-form-intro"
      });

      new Setting(contentEl)
        .setName("显示名称")
        .setDesc("只修改页面标题，不重命名文件夹")
        .addText(text => text
          .setValue(this.values.name)
          .onChange(value => { this.values.name = value; }));

      new Setting(contentEl)
        .setName("订阅状态")
        .addDropdown(dropdown => dropdown
          .addOptions({ "订阅中": "订阅中", "已停用": "已停用" })
          .setValue(this.values.status)
          .onChange(value => { this.values.status = value; }));

      new Setting(contentEl)
        .setName("续费方式")
        .addDropdown(dropdown => dropdown
          .addOptions({ "自动续费": "自动续费", "手动续费": "手动续费" })
          .setValue(this.values.renewal_mode)
          .onChange(value => { this.values.renewal_mode = value; }));

      new Setting(contentEl)
        .setName("订阅周期")
        .addDropdown(dropdown => dropdown
          .addOptions({ "月度": "月度", "季度": "季度", "年度": "年度" })
          .setValue(this.values.cycle)
          .onChange(value => { this.values.cycle = value; }));

      new Setting(contentEl)
        .setName("预计续费价")
        .setDesc("用于估算未来支出，不影响已经记录的实际花费")
        .addText(text => {
          text.inputEl.type = "number";
          text.inputEl.min = "0";
          text.inputEl.step = "0.01";
          text.setValue(this.values.price).onChange(value => { this.values.price = value; });
        });

      new Setting(contentEl)
        .setName("下次扣费 / 到期日期")
        .addText(text => {
          text.inputEl.type = "date";
          text.setValue(this.values.next_date).onChange(value => { this.values.next_date = value; });
        });

      new Setting(contentEl)
        .setName("此前累计支出")
        .setDesc("不含本模块中已经保存的续费记录")
        .addText(text => {
          text.inputEl.type = "number";
          text.inputEl.min = "0";
          text.inputEl.step = "0.01";
          text.setValue(this.values.historical_spend).onChange(value => { this.values.historical_spend = value; });
        });

      new Setting(contentEl)
        .setClass("subscription-form-actions")
        .addButton(button => button
          .setButtonText("取消")
          .onClick(() => this.finish(null)))
        .addButton(button => button
          .setButtonText("保存修改")
          .setCta()
          .onClick(() => {
            const name = this.values.name.trim();
            const price = Number(this.values.price);
            const hasHistoricalSpend = this.values.historical_spend.trim() !== "";
            const historicalSpend = hasHistoricalSpend ? Number(this.values.historical_spend) : null;
            if (!name) {
              new Notice("显示名称不能为空");
              return;
            }
            if (!Number.isFinite(price) || price < 0) {
              new Notice("预计续费价需要填写有效数字");
              return;
            }
            if (!isValidISODate(this.values.next_date)) {
              new Notice("请填写有效的下次扣费或到期日期");
              return;
            }
            if (hasHistoricalSpend && (!Number.isFinite(historicalSpend) || historicalSpend < 0)) {
              new Notice("此前累计支出需要填写有效数字");
              return;
            }
            this.finish({ ...this.values, name, price, historical_spend: historicalSpend });
          }));
    }

    onClose() {
      this.contentEl.empty();
      if (!this.resolved) {
        this.resolved = true;
        resolve(null);
      }
    }
  }

  new EditSubscriptionModal(app).open();
});

const values = await openForm();
if (!values) return;

try {
  await app.fileManager.processFrontMatter(activeFile, data => {
    data.name = values.name;
    data.status = values.status;
    data.renewal_mode = values.renewal_mode;
    data.cycle = values.cycle;
    data.price = values.price;
    data.next_date = values.next_date;
    data.historical_spend = values.historical_spend;
  });
  await new Promise(resolve => setTimeout(resolve, 160));
  await app.commands.executeCommandById("dataview:refresh-views");
  new Notice("订阅信息已更新");
} catch (error) {
  console.error(error);
  new Notice(`保存失败：${error.message ?? error}`);
}
%>
