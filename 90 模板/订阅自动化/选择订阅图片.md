const bootstrap = new Function(await app.vault.adapter.read("90 模板/订阅自动化/scripts/bootstrap.js"))();
const ledger = await bootstrap(app);
const context = app.__subscriptionLedgerContext;
try {
const activeFile = context?.kind === "cover" && context.targetPath
  ? ledger.repository.getItem(context.targetPath).file
  : app.workspace.getActiveFile();
const frontmatter = activeFile ? app.metadataCache.getFileCache(activeFile)?.frontmatter : null;
if (!activeFile || frontmatter?.type !== "subscription") {
  new Notice("请先打开一项订阅的详情页");
  return;
}

const folder = activeFile.parent.path;
const libraryFolder = "90 模板/订阅素材/订阅 Logo";
const extensions = new Set(["png", "jpg", "jpeg", "webp", "gif", "svg", "heic"]);
const localImages = app.vault.getFiles()
  .filter(file => file.parent?.path === folder && extensions.has(file.extension.toLowerCase()))
  .sort((left, right) => right.stat.mtime - left.stat.mtime);
const libraryImages = app.vault.getFiles()
  .filter(file => file.parent?.path === libraryFolder && extensions.has(file.extension.toLowerCase()))
  .sort((left, right) => left.name.localeCompare(right.name, "en"));
const images = [...localImages, ...libraryImages];

if (!images.length) {
  new Notice("没有找到可用图片，请确认内置 Logo 素材是否已经复制到模板目录");
  return;
}

const chooseWithPreview = files => new Promise(resolve => {
  let settled = false;
  class SubscriptionPreviewModal extends tp.obsidian.Modal {
    onOpen() {
      this.modalEl.addClass("subscription-attachment-preview-modal");
      this.contentEl.createEl("h2", { text: "选择订阅图片" });
      this.contentEl.createEl("p", {
        text: `点击图片即可使用。当前订阅文件夹 ${localImages.length} 张，内置 Logo ${libraryImages.length} 张。`,
        cls: "subscription-attachment-preview-help"
      });
      const search = this.contentEl.createEl("input", {
        cls: "subscription-attachment-preview-search",
        attr: { type: "search", placeholder: "搜索文件名，例如 spotify、icloud、bilibili" }
      });
      const grid = this.contentEl.createDiv({ cls: "subscription-attachment-preview-grid" });
      const cards = [];
      for (const file of files) {
        const isLocal = file.parent?.path === folder;
        const card = grid.createEl("button", {
          cls: "subscription-attachment-preview-card",
          attr: { type: "button", "aria-label": `选择 ${file.name}` }
        });
        card.dataset.search = file.name.toLowerCase();
        cards.push(card);
        const preview = card.createDiv({ cls: "subscription-attachment-preview-media" });
        preview.createEl("img", {
          attr: {
            src: app.vault.getResourcePath(file),
            alt: file.name,
            loading: "lazy"
          }
        });
        card.createDiv({
          text: `${isLocal ? "自定义" : "内置"} · ${file.name}`,
          cls: "subscription-attachment-preview-name"
        });
        card.addEventListener("click", () => {
          settled = true;
          resolve(file);
          this.close();
        });
      }
      search.addEventListener("input", () => {
        const keyword = search.value.trim().toLowerCase();
        for (const card of cards) card.hidden = Boolean(keyword) && !card.dataset.search.includes(keyword);
      });
      search.focus();
    }
    onClose() {
      this.contentEl.empty();
      if (!settled) {
        settled = true;
        resolve(null);
      }
    }
  }
  new SubscriptionPreviewModal(app).open();
});

const image = await chooseWithPreview(images);
if (!image) return;

await new Promise(resolve => setTimeout(resolve, 120));

new Notice(`订阅图片已更新：${image.name}`);

} finally {
  if (app.__subscriptionLedgerContext === context) app.__subscriptionLedgerContext = null;
