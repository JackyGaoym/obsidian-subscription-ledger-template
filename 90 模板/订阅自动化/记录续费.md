<%*
const { Modal, Setting } = tp.obsidian;
const HOME_PATH = "30 订阅/订阅主页.md";
const activeFile = app.workspace.getActiveFile();
const activeFrontmatter = activeFile
  ? app.metadataCache.getFileCache(activeFile)?.frontmatter
  : null;

const clearRenewTarget = async () => {
  const homeFile = app.vault.getAbstractFileByPath(HOME_PATH);
  if (!homeFile?.extension) return;
  await app.fileManager.processFrontMatter(homeFile, data => {
    data.renew_target = null;
  });
};

const getSubscriptionName = item => {
  const value = item?.frontmatter?.name;
  return typeof value === "string" && value.trim()
    ? value.trim()
    : item.file.basename;
};

const allSubscriptions = app.vault.getMarkdownFiles()
  .map(file => ({ file, frontmatter: app.metadataCache.getFileCache(file)?.frontmatter }))
  .filter(item => item.file.path.startsWith("30 订阅/项目/")
    && item.frontmatter?.type === "subscription"
    && item.frontmatter?.status !== "已停用")
  .sort((left, right) => getSubscriptionName(left)
    .localeCompare(getSubscriptionName(right), "zh-CN"));

let target = activeFrontmatter?.type === "subscription"
  ? { file: activeFile, frontmatter: activeFrontmatter }
  : null;

if (!target && activeFile?.path === HOME_PATH && activeFrontmatter?.renew_target) {
  const linked = app.vault.getAbstractFileByPath(String(activeFrontmatter.renew_target));
  const linkedFrontmatter = linked?.extension
    ? app.metadataCache.getFileCache(linked)?.frontmatter
    : null;
  if (linked?.extension && linkedFrontmatter?.type === "subscription" && linkedFrontmatter?.status !== "已停用") {
    target = { file: linked, frontmatter: linkedFrontmatter };
  }
  await clearRenewTarget();
}

if (!allSubscriptions.length) {
  new Notice("目前没有可以续费的订阅");
  return;
}

if (target && !allSubscriptions.some(item => item.file.path === target.file.path)) {
  target = null;
}

const pad = value => String(value).padStart(2, "0");
const toISO = date => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
const isValidISODate = value => {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(String(value))) return false;
  const [year, month, day] = String(value).split("-").map(Number);
  const date = new Date(year, month - 1, day);
  return date.getFullYear() === year && date.getMonth() === month - 1 && date.getDate() === day;
};
const parseISO = value => {
  if (!isValidISODate(value)) return null;
  const [year, month, day] = String(value).split("-").map(Number);
  return new Date(year, month - 1, day);
};
const addMonths = (iso, months) => {
  const [year, month, day] = iso.split("-").map(Number);
  const targetDate = new Date(year, month - 1 + months, 1);
  const lastDay = new Date(targetDate.getFullYear(), targetDate.getMonth() + 1, 0).getDate();
  targetDate.setDate(Math.min(day, lastDay));
  return toISO(targetDate);
};

const openForm = initialTarget => new Promise(resolve => {
  class RenewalFormModal extends Modal {
    constructor(app) {
      super(app);
      this.resolved = false;
      this.contextLocked = Boolean(initialTarget);
      const selected = initialTarget ?? allSubscriptions[0];
      this.values = {
        subscription_path: selected.file.path,
        amount: String(selected.frontmatter.price ?? ""),
        payment_date: tp.date.now("YYYY-MM-DD"),
        update_price: false
      };
    }

    finish(value) {
      if (this.resolved) return;
      this.resolved = true;
      resolve(value);
      this.close();
    }

    onOpen() {
      const { contentEl, modalEl } = this;
      modalEl.addClass("subscription-form-modal");
      contentEl.createEl("h2", { text: "记录续费" });
      const introEl = contentEl.createEl("p", {
        text: "选择订阅后，填写这一笔实际支付。",
        cls: "subscription-form-intro"
      });

      const selectedTarget = () => allSubscriptions.find(
        item => item.file.path === this.values.subscription_path
      ) ?? allSubscriptions[0];
      const refreshIntro = () => {
        const selected = selectedTarget();
        const cycle = ["月度", "季度", "年度"].includes(String(selected.frontmatter.cycle))
          ? String(selected.frontmatter.cycle)
          : "月度";
        introEl.setText(`「${getSubscriptionName(selected)}」预计续费价 ${selected.frontmatter.price ?? "—"} 元 / ${cycle.replace("度", "")}；本次实付可以不同。`);
        dateSetting?.setName(String(selected.frontmatter.renewal_mode) === "手动续费" ? "续费日期" : "扣费日期");
      };

      let amountInput;
      let dateSetting;
      const projectSetting = new Setting(contentEl)
        .setName("续费项目");
      if (this.contextLocked) {
        projectSetting
          .setDesc("已从当前订阅进入，本次不会误记到其他项目")
          .setClass("subscription-form-locked-project");
        projectSetting.controlEl.createSpan({
          text: getSubscriptionName(selectedTarget()),
          cls: "subscription-form-static-value"
        });
      } else {
        projectSetting
          .setDesc("只显示当前订阅中的项目")
          .addDropdown(dropdown => {
          for (const item of allSubscriptions) {
            dropdown.addOption(item.file.path, getSubscriptionName(item));
          }
          dropdown
            .setValue(this.values.subscription_path)
            .onChange(value => {
              this.values.subscription_path = value;
              const selected = selectedTarget();
              this.values.amount = String(selected.frontmatter.price ?? "");
              amountInput?.setValue(this.values.amount);
              refreshIntro();
            });
          });
      }

      new Setting(contentEl)
        .setName("本次实际支付")
        .setDesc("优惠或涨价只影响这一笔记录")
        .addText(text => {
          text.inputEl.type = "number";
          text.inputEl.min = "0";
          text.inputEl.step = "0.01";
          amountInput = text;
          text.setValue(this.values.amount).onChange(value => { this.values.amount = value; });
        });

      dateSetting = new Setting(contentEl)
        .setName(String(selectedTarget().frontmatter.renewal_mode) === "手动续费" ? "续费日期" : "扣费日期")
        .addText(text => {
          text.inputEl.type = "date";
          text.setValue(this.values.payment_date).onChange(value => { this.values.payment_date = value; });
        });

      new Setting(contentEl)
        .setName("同步更新预计价")
        .setDesc("只有以后大概率都按这个价格购买时才打开")
        .addToggle(toggle => toggle
          .setValue(false)
          .onChange(value => { this.values.update_price = value; }));

      new Setting(contentEl)
        .setClass("subscription-form-actions")
        .addButton(button => button
          .setButtonText("取消")
          .onClick(() => this.finish(null)))
        .addButton(button => button
          .setButtonText("记录续费")
          .setCta()
          .onClick(() => {
            const amount = Number(this.values.amount);
            if (!Number.isFinite(amount) || amount < 0) {
              new Notice("续费金额需要填写有效数字");
              return;
            }
            if (!isValidISODate(this.values.payment_date)) {
              new Notice("请填写有效的支付日期");
              return;
            }
            this.finish({ ...this.values, amount, target: selectedTarget() });
          }));

      refreshIntro();
    }

    onClose() {
      this.contentEl.empty();
      if (!this.resolved) {
        this.resolved = true;
        resolve(null);
      }
    }
  }

  new RenewalFormModal(app).open();
});

const values = await openForm(target);
if (!values) {
  await clearRenewTarget();
  return;
}

target = values.target;
const subscription = target.frontmatter;
const name = getSubscriptionName(target);
const cycle = ["月度", "季度", "年度"].includes(String(subscription.cycle))
  ? String(subscription.cycle)
  : "月度";

const cycleMonths = { "月度": 1, "季度": 3, "年度": 12 }[cycle];
const oldDate = String(subscription.next_date ?? "").slice(0, 10);
const oldDateValue = parseISO(oldDate);
const paymentDateValue = parseISO(values.payment_date);
const baseDate = oldDateValue && paymentDateValue && oldDateValue > paymentDateValue
  ? oldDate
  : values.payment_date;
const nextDate = addMonths(baseDate, cycleMonths);

const recordRoot = "30 订阅/_续费记录";
const yearFolder = `${recordRoot}/${values.payment_date.slice(0, 4)}`;
if (!app.vault.getAbstractFileByPath(recordRoot)) await app.vault.createFolder(recordRoot);
if (!app.vault.getAbstractFileByPath(yearFolder)) await app.vault.createFolder(yearFolder);

const safeName = name.replace(/[\\/:*?"<>|]/g, "-");
const stamp = tp.date.now("HHmmss");
let recordPath = `${yearFolder}/${safeName} · ${values.payment_date} · ${stamp}.md`;
let suffix = 2;
while (app.vault.getAbstractFileByPath(recordPath)) {
  recordPath = `${yearFolder}/${safeName} · ${values.payment_date} · ${stamp}-${suffix}.md`;
  suffix += 1;
}

const recordContent = [
  "---",
  "type: subscription_payment",
  `subscription: ${JSON.stringify(`[[${target.file.path}|${name}]]`)}`,
  `payment_date: ${values.payment_date}`,
  `amount: ${values.amount}`,
  `cycle: ${JSON.stringify(cycle)}`,
  `standard_price_before: ${Number(subscription.price) || 0}`,
  `price_updated: ${values.update_price}`,
  `previous_date: ${oldDate || ""}`,
  `next_date: ${nextDate}`,
  "---",
  ""
].join("\n");

try {
  await app.vault.create(recordPath, recordContent);
  await app.fileManager.processFrontMatter(target.file, data => {
    data.next_date = nextDate;
    if (values.update_price) data.price = values.amount;
  });
  await clearRenewTarget();
  await new Promise(resolve => setTimeout(resolve, 160));
  await app.commands.executeCommandById("dataview:refresh-views");
  new Notice(`「${name}」续费已记录，下次日期为 ${nextDate}`);
  if (activeFile?.path !== HOME_PATH) await app.workspace.getLeaf(false).openFile(target.file);
} catch (error) {
  console.error(error);
  await clearRenewTarget();
  new Notice(`记录失败：${error.message ?? error}`);
}
%>
