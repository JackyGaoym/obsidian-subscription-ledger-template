const pad = value => String(value).padStart(2, "0");
const toISO = date => `${date.getFullYear()}-${pad(date.getMonth() + 1)}-${pad(date.getDate())}`;
const offsetISO = days => {
  const date = new Date(today);
  date.setDate(date.getDate() + days);
  return toISO(date);
};
const normalizePath = value => value?.path
  ?? String(value ?? "").replace(/^\[\[|\]\]$/g, "").split("|")[0];
const toDate = value => {
  if (!value) return null;
  if (value instanceof Date) return value;
  if (typeof value.toJSDate === "function") return value.toJSDate();
  const date = new Date(`${String(value).slice(0, 10)}T00:00:00`);
  return Number.isNaN(date.getTime()) ? null : date;
};
const formatDate = value => {
  const date = toDate(value);
  if (!date) return "未记录";
  return `${date.getFullYear()}.${pad(date.getMonth() + 1)}.${pad(date.getDate())}`;
};
const formatShortDate = value => {
  const date = toDate(value);
  return date ? `${pad(date.getMonth() + 1)}.${pad(date.getDate())}` : "—";
};
const formatMoney = (value, decimals = null) => {
  const amount = Number(value);
  if (!Number.isFinite(amount)) return "—";
  const digits = decimals ?? (Number.isInteger(amount) ? 0 : 2);
  return `¥${amount.toLocaleString("zh-CN", {
    minimumFractionDigits: digits,
    maximumFractionDigits: digits
  })}`;
};
const amountOrZero = value => {
  const amount = Number(value);
  return Number.isFinite(amount) && amount >= 0 ? amount : 0;
};
const cycleMonths = cycle => ({ "月度": 1, "季度": 3, "年度": 12 }[String(cycle)] ?? 1);
const monthlyCost = item => amountOrZero(item.price) / cycleMonths(item.cycle);
const dayDistance = value => {
  const date = toDate(value);
  return date ? Math.ceil((date - today) / 86400000) : null;
};
const eventCopy = item => {
  const days = dayDistance(item.next_date);
  const action = item.renewal_mode === "自动续费" ? "扣费" : "到期";
  if (days === null) return `日期未记录`;
  if (days === 0) return `今天${action}`;
  if (days > 0) return `${days}天后${action}`;
  return item.renewal_mode === "自动续费" ? `已过扣费日${Math.abs(days)}天` : `已到期${Math.abs(days)}天`;
};
const dateLabel = item => item.renewal_mode === "自动续费" ? "下次扣费" : "到期";

const paymentPages = dv.pages('"30 订阅/_续费记录"')
  .where(page => page.type === "subscription_payment" && page.voided !== true)
  .array();
const realPages = dv.pages('"30 订阅/项目"')

const subscriptions = previewMode
  ? demoSubscriptions
  : realPages.map(page => {
      const path = page.file.path;
      const matchingPayments = paymentPages
        .filter(payment => normalizePath(payment.subscription) === path);
      const historicalSpend = amountOrZero(page.historical_spend);
      const recordedSpend = matchingPayments
        .reduce((total, payment) => total + amountOrZero(payment.amount), historicalSpend);
      return {
        name: page.name || page.file.name,
        path,
        status: String(page.status) === "已停用" ? "已停用" : "订阅中",

        cycle: ["月度", "季度", "年度"].includes(String(page.cycle)) ? String(page.cycle) : "月度",
        price: Number(page.price),
        next_date: page.next_date,
        recorded_spend: recordedSpend,
        has_spend_data: matchingPayments.length > 0 || historicalSpend > 0,
        cover: normalizePath(page.cover)
      };
    });

  const link = parent.createEl("a", { text, cls: `internal-link ${className}`.trim() });

const launchRenew = async item => {
  if (item && !item.path) {
    new Notice("这是示例订阅；添加真实订阅后即可记录续费");
    return;
  }
  if (item?.path) {
    const homeFile = app.vault.getAbstractFileByPath(HOME_PATH);
    if (homeFile?.extension) {
      await app.fileManager.processFrontMatter(homeFile, data => {
        data.renew_target = item.path;
      });
      await new Promise(resolve => setTimeout(resolve, 160));
    }
  }
  const trigger = findTrigger("renew");
  if (trigger instanceof HTMLElement) trigger.click();
  else new Notice("记录续费表单暂时没有加载，请重新打开订阅主页");

      const selected = candidate === button;
      candidate.classList.toggle("is-active", selected);
      candidate.setAttr("aria-pressed", selected ? "true" : "false");

    let visible = 0;
    for (const row of rows) {
      const days = row.dataset.days === "" ? null : Number(row.dataset.days);
      const show = filter === "all"
        || (filter === "due30" && row.dataset.status === "订阅中" && days !== null && days >= 0 && days <= 30)
        || (filter === "overdue" && row.dataset.status === "订阅中" && days !== null && days < 0)
        || (filter === "auto" && row.dataset.mode === "自动续费" && row.dataset.status === "订阅中")
        || (filter === "manual" && row.dataset.mode === "手动续费" && row.dataset.status === "订阅中")
        || (filter === "stopped" && row.dataset.status === "已停用");
      row.hidden = !show;
      if (show) visible += 1;
    }
    empty.hidden = visible > 0;
  });
}
